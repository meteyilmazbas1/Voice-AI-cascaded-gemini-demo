527:// Play raw PCM audio (from Gemini — audio/pcm;rate=24000)
528:async function playPCMChunks(chunks) {
529:  try {
530:    const ctx = getAudioContext();
531:    const sampleRate = 24000;
532:    const allBytes = chunks.map(b64 => {
533:      const bin = atob(b64);
534:      const arr = new Uint8Array(bin.length);
535:      for (let i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i);
536:      return arr;
537:    });
538:    const totalLen = allBytes.reduce((sum, a) => sum + a.length, 0);
539:    const combined = new Uint8Array(totalLen);
540:    let offset = 0;
541:    for (const arr of allBytes) { combined.set(arr, offset); offset += arr.length; }
543:    // PCM 16-bit little-endian → float32
544:    const int16 = new Int16Array(combined.buffer);
545:    const float32 = new Float32Array(int16.length);
546:    for (let i = 0; i < int16.length; i++) float32[i] = int16[i] / 32768.0;
548:    const audioBuffer = ctx.createBuffer(1, float32.length, sampleRate);
549:    audioBuffer.getChannelData(0).set(float32);
550:    const src = ctx.createBufferSource();
551:    src.buffer = audioBuffer;
552:    src.connect(ctx.destination);
553:    src.start();
554:  } catch(e) { console.warn('[PCM] Decode error:', e.message); }
555:}
